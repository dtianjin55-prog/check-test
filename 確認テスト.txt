const app = document.getElementById("app");
let timerId = null;

function renderQuestion(){
  const d = QA[idx];
  app.innerHTML = `
    <div class="card">
      <h2>Q${d.no}. ${d.q}</h2>
      <p><b>分野：</b>${d.cat} ／ <b>難度：</b>${d.level}</p>
      <p><b>深掘り質問：</b>${d.probe}</p>
      <div class="timer">回答時間：${d.time}秒</div>
      <button class="btn primary" onclick="startAnswer()">回答を開始</button>
    </div>
  `;
}

function startAnswer(){
  const d = QA[idx];
  let t = d.time;
  app.innerHTML = `
    <div class="card">
      <h2>回答中</h2>
      <p>声に出して、結論から回答してみてください。</p>
      <div class="timer" id="tm">${t} 秒</div>
    </div>
  `;
  timerId = setInterval(()=>{
    t--;
    const el = document.getElementById("tm");
    if(!el){clearInterval(timerId);return;}
    el.textContent = t + " 秒";
    if(t<=0){
      clearInterval(timerId);
      renderAnswer();
    }
  },1000);
}

}
